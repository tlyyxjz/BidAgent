```python
# tests/test_extractor.py 追加
class TestMarkdownFenceResponse:
    def test_parse_response_wrapped_in_json_fence(self):
        """约束 #42：LLM client 必须处理 Markdown 代码块。
        当前 extractor.py 直接 json.loads(content)，```json 包裹必抛异常 → 红测试。"""
        # GIVEN：LLM 返回 ```json 包裹的合法响应（deepseek 在未严格遵循 json_object 时常见）
        raw_content = '```json\n{"fields": [{"field_name": "project_identifier", "field_status": "present", "raw_value": "ZFCG-2026-001"}]}\n```'
        # WHEN：走 call_extraction_llm 的 content 处理路径（此处直接测解析前置步骤）
        import json as _json
        # THEN：应能剥离 fence 后解析成功；当前实现此处抛 json.JSONDecodeError
        stripped = raw_content.strip()
        if stripped.startswith("```"):
            stripped = stripped.split("\n", 1)[1].rsplit("```", 1)[0]
        data = _json.loads(stripped)
        result = parse_extraction_response(data, "test", 0)
        assert result.fields[0].raw_value == "ZFCG-2026-001"
```

**P0-2｜#15：非法 cron 表达式不得导致崩溃/误触发**

```python
# tests/test_enterprise_quality.py 或新建 tests/test_scheduler_utils.py
from datetime import datetime, timezone
from app.scheduler.utils import is_cron_due

class TestCronValidation:
    def test_is_cron_due_invalid_expr_no_crash(self):
        """约束 #15：非法 frequency_cron 必须在触发前被识别。"""
        # GIVEN：非法 cron 表达式 + 有历史运行记录
        now = datetime(2026, 7, 27, 12, 0, tzinfo=timezone.utc)
        last = datetime(2026, 7, 27, 11, 0, tzinfo=timezone.utc)
        # WHEN：调用 is_cron_due（croniter 对非法表达式抛 ValueError）
        # THEN：不得向外抛异常；应返回 False 或抛出带中文提示的 ValueError（二选一，锁定行为）
        try:
            result = is_cron_due("not_a_cron", last, now)
            assert result is False, "非法 cron 应返回 False（不得误触发）"
        except ValueError as exc:
            assert "cron" in str(exc).lower(), "若抛异常必须含 cron 语义提示"

    def test_is_cron_due_never_ran_returns_false(self):
        """约束 #15 边界：S-6 修复——从未运行的订阅不得立即触发。"""
        # GIVEN：合法 cron + last_run=None
        # WHEN / THEN
        assert is_cron_due("0 9 * * *", None, datetime.now(timezone.utc)) is False

**P0-3｜#26：BrowserPool 部分启动失败必须清理已启动 slot**

```python
# tests/test_browser_pool.py 追加（修复该文件预存在超时后一并解除 --ignore）
from unittest.mock import AsyncMock, patch
import pytest
from app.core.browser_pool import BrowserPool

class TestPartialStartupCleanup:
    @pytest.mark.asyncio
    async def test_second_slot_failure_closes_first_slot(self):
        """约束 #26：第 2 个 slot 启动失败时，第 1 个已启动 slot 必须被 close。"""
        # GIVEN：size=2 的池，mock playwright 使第 2 次 browser 启动抛异常
        pool = BrowserPool(size=2, acquire_timeout=1)
        closed_slots = []
        original_close = pool.close

        async def spy_close():
            closed_slots.append("close_called")
            await original_close()

        # WHEN：start() 在第 2 个 slot 处失败
        with patch.object(pool, "close", side_effect=spy_close):
            with patch("app.core.browser_pool.async_playwright") as mock_pw:
