# app/graph/prototype.py
from neo4j import GraphDatabase
from typing import Any

class TenderGraphRepository:
    def __init__(self, uri: str, user: str, password: str):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))

    def close(self):
        self.driver.close()

    def create_tender(self, tender: dict[str, Any]) -> str:
        """创建招标项目节点 + 关联机构/地区/类别节点"""
        with self.driver.session() as session:
            result = session.run(
                """
                MERGE (t:Tender {bid_number: $bid_number})
                SET t.project_name = $project_name,
                    t.budget = $budget,
                    t.publish_time = $publish_time,
                    t.deadline = $deadline
                MERGE (o:Organization {name: $tender_org})
                MERGE (r:Region {name: $location})
                MERGE (t)-[:PUBLISHED_BY]->(o)
                MERGE (t)-[:LOCATED_IN]->(r)
                RETURN t.bid_number
                """,
                **tender
            )
            return result.single()["t.bid_number"]

    def find_related_tenders(self, bid_number: str, limit: int = 10) -> list[dict]:
        """查询关联招标项目（同地区同类别）"""
        with self.driver.session() as session:
            result = session.run(
                """
                MATCH (t:Tender {bid_number: $bid_number})-[:LOCATED_IN]->(r:Region)<-[:LOCATED_IN]-(related:Tender)
                WHERE related.bid_number <> $bid_number
                RETURN related.bid_number as bid_number,
                       related.project_name as project_name,
                       related.budget as budget
                LIMIT $limit
                """,
                bid_number=bid_number, limit=limit
            )
            return [dict(record) for record in result]

    def get_region_statistics(self) -> list[dict]:
        """地区招标数量统计"""
        with self.driver.session() as session:
            result = session.run(
                """
                MATCH (t:Tender)-[:LOCATED_IN]->(r:Region)
                RETURN r.name as region, count(t) as tender_count, sum(t.budget) as total_budget
                ORDER BY tender_count DESC
                """
            )
            return [dict(record) for record in result]

- 加分项：多 Agent 协作完成"采集 → 清洗 → 校验 → 推送"完整链路
- 答辩亮点：展示 LangGraph 风格的状态机

## 2. Agent 角色
- IntentParser Agent：解析用户自然语言 → 5 槽位 ParsedFilters
- Collector Agent：根据 ParsedFilters 调度多平台采集
- Cleaner Agent：SimHash 去重 + 字段归一化
- Validator Agent：反幻觉校验 + 输出可信度评分
- Reporter Agent：生成 Word 报告 + 命名规则
- Pusher Agent：增量推送（邮件/Webhook）

## 3. 状态机流转
```
[用户输入] → IntentParser → [ParsedFilters]
                              ↓
                          Collector → [ScrapeResult]
                              ↓
                          Cleaner → [DedupedItems]
                              ↓
                          Validator → [ValidatedItems + 置信度]
                              ↓
                          Reporter → [Word 报告路径]
                              ↓
                          Pusher → [推送结果]
```

## 4. 代码演示（基于 app/agents/coordinator.py）
```python
# 演示脚本：自然语言 → 完整报告
user_input = "帮我查上海地区最近7天预算100万以上的招标信息"
result = await coordinator.run(user_input)
print(f"采集到 {result['tender_count']} 条，报告: {result['report_path']}")
```

## 5. 与传统 ETL 的区别
- 传统：固定流程，单点失败导致全链路失败
- Agent：状态机驱动，每个 Agent 独立失败重试，部分成功语义
- 答辩亮点：SAVEPOINT 事务 + Agent 状态机 = 企业级健壮性
```
